```powershell
python -B phase8/build_curriculum.py
python -B phase8/build_curriculum.py --check
dotnet build phase2/src/Masar.Api/Masar.Api.csproj --output phase8/.local/api
dotnet phase8/.local/api/Masar.Api.dll --verify-curriculum ((Resolve-Path 'phase8/dist/curriculum.json').Path)
```
